using System.Collections;
using System.Collections.ObjectModel;
using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using CardGame.Wpf.Controls;
using CardGame.Wpf.Presentation;

internal static class Program
{
    [STAThread]
    private static int Main(string[] args)
    {
        var output = args[0];
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary
        {
            Source = new Uri("pack://application:,,,/CardGame.Wpf;component/Themes/TableTheme.xaml")
        });
        var checks = Assembly.Load("CardGame.Wpf.Tests").GetType("BattleTextReadabilityChecks", true)!;
        var scene = checks.GetMethod("Scene", BindingFlags.Static | BindingFlags.NonPublic)!.Invoke(null, null)!;
        object Part(string name) => scene.GetType().GetField(name)!.GetValue(scene)!;
        var root = (Canvas)Part("Item1");
        var layer = (BattleFeedbackLayer)Part("Item2");
        var cues = (ObservableCollection<BattleCue>)Part("Item3");
        var clock = Part("Item4");
        try
        {
            layer.IsPromptVisible = true;
            long sequence = 10;
            foreach (var seat in new[] { 0, 1, 2 })
                foreach (var kind in new[] { BattleCueKind.Damage, BattleCueKind.Recovery, BattleCueKind.Damage })
                    cues.Add(new(sequence++, kind, 0, [seat], kind == BattleCueKind.Damage ? "−12" : "+3",
                        seat == 0 ? "关羽" : seat == 1 ? "曹操" : "黄盖"));
            clock.GetType().GetMethod("Advance")!.Invoke(clock, [.17]);
            Capture(output + "-before.png");
            var initialBounds = NumberBounds();
            if (initialBounds.Length != 9) throw new InvalidOperationException("The fixed drawing scene must show all 9 results.");
            var render = (Action<DrawingContext>)typeof(BattleFeedbackLayer)
                .GetMethod("OnRender", BindingFlags.Instance | BindingFlags.NonPublic)!.CreateDelegate(typeof(Action<DrawingContext>), layer);
            var visual = new DrawingVisual();
            void Frame()
            {
                using var dc = visual.RenderOpen();
                render(dc);
            }
            for (var i = 0; i < 12; i++) Frame();
            var samples = new List<object>();
            const int frames = 128;
            for (var sample = 0; sample < 3; sample++)
            {
                GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
                var startAllocated = GC.GetAllocatedBytesForCurrentThread();
                var startCpu = ThreadCpuTicks();
                var watch = Stopwatch.StartNew();
                for (var frame = 0; frame < frames; frame++) Frame();
                watch.Stop();
                var cpuMs = (ThreadCpuTicks() - startCpu) / (double)TimeSpan.TicksPerMillisecond;
                var allocated = GC.GetAllocatedBytesForCurrentThread() - startAllocated;
                samples.Add(new { Frames = frames, ElapsedMs = watch.Elapsed.TotalMilliseconds, UiThreadCpuMs = cpuMs,
                    ManagedAllocatedBytes = allocated, ManagedBytesPerFrame = allocated / (double)frames });
            }
            Capture(output + "-after.png");
            var finalBounds = NumberBounds();
            if (JsonSerializer.Serialize(initialBounds) != JsonSerializer.Serialize(finalBounds))
                throw new InvalidOperationException("Drawing measurement changed fixed-clock number placement.");
            var result = new { ClockSeconds = .17, CueCount = 9, WarmupFrames = 12,
                Measurement = "Complete BattleFeedbackLayer.OnRender plus DrawingVisual.RenderOpen on the UI thread, no PNG encoding inside timed loops",
                Samples = samples, NumberBounds = finalBounds, ActiveEffects = layer.ActiveEffectCount };
            File.WriteAllText(output + ".json", JsonSerializer.Serialize(result, new JsonSerializerOptions { WriteIndented = true }));
            Console.WriteLine(JsonSerializer.Serialize(new { Samples = samples, NumberCount = finalBounds.Length, result.ActiveEffects }));
            return 0;
        }
        finally
        {
            layer.MotionEnabled = false;
            cues.Clear();
            root.Children.Clear();
            app.Shutdown();
        }

        void Capture(string path)
        {
            layer.InvalidateVisual();
            root.Measure(new Size(850, 500)); root.Arrange(new Rect(0, 0, 850, 500)); root.UpdateLayout();
            var bitmap = new RenderTargetBitmap(850, 500, 96, 96, PixelFormats.Pbgra32); bitmap.Render(root);
            var encoder = new PngBitmapEncoder(); encoder.Frames.Add(BitmapFrame.Create(bitmap));
            using var stream = File.Create(path); encoder.Save(stream);
        }
        object[] NumberBounds()
        {
            var entries = (IEnumerable)checks.GetMethod("Numbers", BindingFlags.Static | BindingFlags.NonPublic)!.Invoke(null, [layer])!;
            var outputBounds = new List<object>();
            foreach (var entry in entries)
            {
                var type = entry.GetType();
                var bounds = (Rect)type.GetField("Item2")!.GetValue(entry)!;
                var origin = (double)type.GetField("Item3")!.GetValue(entry)!;
                outputBounds.Add(new { Color = type.GetField("Item1")!.GetValue(entry)!.ToString(),
                    X = bounds.X, Y = bounds.Y, Width = bounds.Width, Height = bounds.Height, OriginX = origin });
            }
            return outputBounds.ToArray();
        }
    }

    private static long ThreadCpuTicks()
    {
        if (!GetThreadTimes(GetCurrentThread(), out _, out _, out var kernel, out var user))
            throw new InvalidOperationException("Unable to measure the actual drawing thread CPU time.");
        return kernel + user;
    }
    [DllImport("kernel32.dll")] private static extern IntPtr GetCurrentThread();
    [DllImport("kernel32.dll")] private static extern bool GetThreadTimes(IntPtr thread,
        out long creation, out long exit, out long kernel, out long user);
}
