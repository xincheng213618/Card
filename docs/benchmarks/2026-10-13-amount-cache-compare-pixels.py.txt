import json
import sys
from pathlib import Path
from PIL import Image, ImageChops

base = Path(sys.argv[1])
pairs = [
    ("consecutive before-after", f"{base}-before-battle-text-consecutive.png", f"{base}-after-battle-text-consecutive.png"),
    ("crowd before-after", f"{base}-before-battle-text-edge-crowd.png", f"{base}-after-battle-text-edge-crowd.png"),
    ("drawing scene before-after", f"{base}-before-drawing-1-before.png", f"{base}-after-drawing-1-before.png"),
    ("before process agreement", f"{base}-before-drawing-1-before.png", f"{base}-before-drawing-2-before.png"),
    ("after process agreement", f"{base}-after-drawing-1-before.png", f"{base}-after-drawing-2-before.png"),
]
for variant in ("before", "after"):
    for run in (1, 2):
        pairs.append((f"{variant} run{run} measurement stable", f"{base}-{variant}-drawing-{run}-before.png", f"{base}-{variant}-drawing-{run}-after.png"))

results = []
for name, left, right in pairs:
    with Image.open(left) as left_file, Image.open(right) as right_file:
        a, b = left_file.convert("RGBA"), right_file.convert("RGBA")
        if a.size != b.size:
            raise RuntimeError(f"Image dimensions changed: {name}")
        diff = ImageChops.difference(a, b)
        r, g, blue, alpha = diff.split()
        mask = ImageChops.lighter(ImageChops.lighter(r, g), ImageChops.lighter(blue, alpha))
        results.append({"Name": name, "Left": str(left), "Right": str(right), "Size": a.size,
                        "ChangedPixels": sum(value != 0 for value in mask.tobytes()),
                        "MaxChannelDelta": max(diff.tobytes(), default=0), "DifferenceBounds": mask.getbbox()})
report = {"Mode": "Full decoded RGBA pixel comparison, including alpha", "Comparisons": results,
          "AllEqual": all(result["ChangedPixels"] == 0 for result in results)}
Path(f"{base}-pixels.json").write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
print(json.dumps(report, ensure_ascii=False))
sys.exit(0 if report["AllEqual"] else 1)
