using System.Diagnostics;
using System.Reflection;
using System.Text.Json;
using CardGame.Core;
using CardGame.Content.Standard;

var assembly = Assembly.Load("CardGame.Wpf.Tests");
var fixture = assembly.GetType("StableHandResponseFixture+Package", throwOnError: true)!;
var constructor = fixture.GetConstructors(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance).Single();
var owner = assembly.GetType("StableHandResponseFixture", throwOnError: true)!;
var keyType = owner.GetNestedType("RegistryKey", BindingFlags.NonPublic)!;
var acquire = owner.GetMethod("GetRegistry", BindingFlags.NonPublic | BindingFlags.Static)!;
var hashes = new Dictionary<string, string>();
ContentRegistry Build(bool zhefu)
{
    var package = (IGameContentPackage)constructor.Invoke([DecisionKind.RespondDodge, false, false, zhefu, false, !zhefu, false]);
    var registry = ContentRegistry.Build(new StandardContentPackage(), package);
    var key = zhefu ? "Zhefu" : "Wushuang";
    if (hashes.TryGetValue(key, out var prior) && prior != registry.ContentHash)
        throw new InvalidOperationException("Fresh fixture content changed during acquisition.");
    hashes[key] = registry.ContentHash;
    return registry;
}
ContentRegistry Reuse(bool zhefu)
{
    var key = Activator.CreateInstance(keyType, BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance,
        binder: null, args: [DecisionKind.RespondDodge, false, false, zhefu, false, !zhefu, false], culture: null)!;
    return (ContentRegistry)acquire.Invoke(null, [key])!;
}
void Validate(ContentRegistry registry, bool zhefu)
{
    if (registry.ContentHash != hashes[zhefu ? "Zhefu" : "Wushuang"] ||
        registry.Modes["identity:stable-hand-responses"].GeneralPoolIds!.Count != 4 ||
        registry.Decks["fixture:response-deck"].PhysicalCards!.Count != 80)
        throw new InvalidOperationException("The actual response registry lost its definitions or content hash.");
}
for (var warmup = 0; warmup < 4; warmup++)
{
    Build(true); Build(false); Validate(Reuse(true), true); Validate(Reuse(false), false);
}
if (!ReferenceEquals(Reuse(true), Reuse(true)) || !ReferenceEquals(Reuse(false), Reuse(false)) ||
    ReferenceEquals(Reuse(true), Reuse(false)))
    throw new InvalidOperationException("The actual registry cache must reuse identical keys and separate distinct keys.");
var samples = new List<object>();
for (var sample = 0; sample < 6; sample++)
{
    foreach (var reuse in sample % 2 == 0 ? new[] { false, true } : new[] { true, false })
    {
        GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
        var allocated = GC.GetAllocatedBytesForCurrentThread();
        var started = Stopwatch.GetTimestamp();
        foreach (var zhefu in new[] { true, false, false, true })
        {
            var registry = reuse ? Reuse(zhefu) : Build(zhefu);
            Validate(registry, zhefu);
            GC.KeepAlive(registry);
        }
        samples.Add(new { Sample = sample, Reuse = reuse, Registries = 4,
            ElapsedMs = Stopwatch.GetElapsedTime(started).TotalMilliseconds,
            ManagedBytes = GC.GetAllocatedBytesForCurrentThread() - allocated });
    }
}
Console.WriteLine(JsonSerializer.Serialize(new { Samples = samples, ContentHashes = hashes,
    Scope = "Actual private response Package.Register plus fresh ContentRegistry.Build versus actual GetRegistry warm reuse; same two immutable definitions and four flag acquisitions; alternating pair order; reflection overhead included; no game/replay/render or cold startup timing" }));
